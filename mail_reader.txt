function isEnd()
	
	return input:sub(-8) == ".-.-...-"
	
end

function toString(input)
  local s = ""
  local len = #input
  for i = 1, len, 8 do
    local chunk = input:sub(i, i + 7)
    local code = codes[chunk] or "_"
    s = s .. code
  end
  return s
end
