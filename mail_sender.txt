speed = 5;

tick = false

function sendChar(char)
    local c = codes[char];
    if not c then return end
    
    -- ГАРАНТИРУЕМ, ЧТО left ИЗМЕНИТСЯ ДЛЯ ПЕРВОГО БИТА
    rs.setOutput("left", true)
    os.sleep(0.01)
    write(char, ": ")
    for i = 1, #c do
        -- УСТАНАВЛИВАЕМ ДАННЫЕ
        rs.setOutput("back", c:sub(i,i) == "-")
        if c:sub(i,i) == "-" then write("-") else write(".") end
        -- ПОСЫЛАЕМ ТАКТ (ИЗМЕНЕНИЕ left)
        rs.setOutput("left", tick)
		tick = not tick
        os.sleep(1 / speed)
    end
	print()
end

function sendString(str)
	
	os.sleep(0.1)
	
	for i = 1, #str do
			sendChar(str:sub(i,i))
	end
	sendChar("&")
	
	rs.setOutput("left", tick)
	tick = not tick
	rs.setOutput("left", false)
	rs.setOutput("back", false)
end

rs.setOutput("left", false)
rs.setOutput("back", false)
sendString("Hello!")
